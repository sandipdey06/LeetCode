import java.util.*;
class Solution {
    public int minSwaps(int[] nums) {
        int n=nums.length;
        int[][] arr=new int[n][2];  //for all cases including 2d arrays
        for (int i=0; i<n; i++) {
            int x=nums[i],sum=0;
            while (x>0){
                sum+=x%10;
                x/=10;
            }
            arr[i][0]=nums[i];
            arr[i][1]=sum;
        }
        Arrays.sort(arr,(a,b) ->
            a[1]!=b[1] ? a[1] - b[1] : a[0] - b[0]
        );
        int count=0;
        for (int i=0; i<n; i++) {
            if (nums[i]!=arr[i][0]) {
                int j=i+1;
                while (nums[j]!=arr[i][0]) {
                    j++;
                }
                int temp=nums[i];
                nums[i]=nums[j];
                nums[j]=temp;
                count++;
            }
        }
        return count;
    }
}
