class Solution {
    public int[] maxDepthAfterSplit(String seq) {
        int len = seq.length();
        int[] a = new int[len];
        int depth = 0;

        for (int i = 0; i < len; i++) {
            if (seq.charAt(i) == '(') {
                a[i] = depth % 2;
                depth++;
            } else {
                depth--;
                a[i] = depth % 2;
            }
        }

        return a;
    }
}
