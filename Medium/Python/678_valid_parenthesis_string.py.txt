class Solution(object):
    def checkValidString(self, s):
        # for i in range(len(s)):
        #     if '(' and ')' not in s:
        #         return False
        #     # elif ')' 
        # return True
        low = high = 0
        for c in s:
            if c == '(':
                low += 1
                high += 1
            elif c == ')':
                low -= 1
                high -= 1
            else:  # *
                low -= 1
                high += 1

            if high < 0:
                return False
            low = max(low, 0)
        return low == 0
