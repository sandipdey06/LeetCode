
class Solution(object):
    def multiply(self, num1, num2):
        if "0" in (num1, num2):
            return "0"

        res = [0] * (len(num1) + len(num2))

        for i in range(len(num1)-1, -1, -1):
            for j in range(len(num2)-1, -1, -1):
                mul = int(num1[i]) * int(num2[j])
                s = mul + res[i+j+1]

                res[i+j+1] = s % 10
                res[i+j] += s // 10

        return ''.join(map(str, res)).lstrip('0')
