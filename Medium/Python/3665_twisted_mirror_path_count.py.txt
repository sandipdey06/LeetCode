class Solution(object):
    def uniquePaths(self, grid):
        m = len(grid)
        n = len(grid[0])
        mod = 10**9 + 7

        right = [[0] * n for _ in range(m)]
        down = [[0] * n for _ in range(m)]

        right[0][0] = 1

        for i in range(m):
            for j in range(n):

                if i == m - 1 and j == n - 1:
                    continue

                if grid[i][j] == 1:
                    # Mirror reflects the direction
                    if i + 1 < m:
                        down[i + 1][j] += right[i][j]
                        down[i + 1][j] %= mod

                    if j + 1 < n:
                        right[i][j + 1] += down[i][j]
                        right[i][j + 1] %= mod

                else:
                    ways = (right[i][j] + down[i][j]) % mod

                    if j + 1 < n:
                        right[i][j + 1] += ways
                        right[i][j + 1] %= mod

                    if i + 1 < m:
                        down[i + 1][j] += ways
                        down[i + 1][j] %= mod

        return (right[m-1][n-1] + down[m-1][n-1]) % mod
