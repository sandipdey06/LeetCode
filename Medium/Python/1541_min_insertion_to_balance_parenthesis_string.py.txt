class Solution(object):
    def minInsertions(self, s):
        #for every '('=='))' 
        left=0
        right=0
        ans=0
        for i in s:
            if i=='(':
                left+=2
                if left%2:
                    ans+=1
                    left-=1
            else:
                left-=1
                if left<0:
                    ans+=1
                    left=1
        return ans+left
