class ProductOfNumbers:
    def __init__(self):
        self.list=[1]
    def add(self,num):
        if num==0:
            self.list=[1]
        else:
            self.list.append(self.list[-1]*num)
    def getProduct(self,k):
        n=len(self.list)
        if k>=n:
            return 0
        return self.list[-1]//self.list[n-k-1]
