class Solution:
    def divide(self,dividend,divisor):
        a=abs(dividend)
        b=abs(divisor)
        ans=0
        while a>=b:
            temp=b
            count=1
            while a>=(temp<<1):
                temp<<=1
                count<<=1
            a-=temp
            ans+=count
        if (dividend<0)!=(divisor<0):
            ans=-ans
        return min(max(ans,-2**31),2**31-1)
