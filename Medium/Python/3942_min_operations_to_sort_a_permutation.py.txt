class Solution(object):
    def minOperations(self, nums):
        n = len(nums)
        zero = nums.index(0)

        inc = True
        dec = True

        for i in range(1, n):
            x = nums[(zero + i) % n]

            if x != i:
                inc = False

            if x != n - i:
                dec = False

        ans = float('inf')

        if inc:
            ans = min(ans, zero, n - zero + 2)

        if dec:
            ans = min(ans, zero + 2, n - zero)

        return -1 if ans == float('inf') else ans
