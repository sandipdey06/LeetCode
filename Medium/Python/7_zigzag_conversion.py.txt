class Solution(object):
    def convert(self, s, numRows):
        if numRows==1 or numRows>=len(s):
            return s
        rows=[""]*numRows
        r=0
        direction=1
        for ch in s:
            rows[r]+=ch
            if r==0:
                direction=1
            elif r==numRows-1:
                direction=-1
            r+=direction
        return "".join(rows)
