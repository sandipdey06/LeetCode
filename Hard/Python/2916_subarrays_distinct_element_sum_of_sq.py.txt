class Solution:
    def sumCounts(self, nums):
        n = len(nums)
        MOD = 10**9 + 7

        sum1 = [0] * (4 * n)
        sum2 = [0] * (4 * n)
        lazy = [0] * (4 * n)

        def push(node, left, right):
            if lazy[node] == 0 or left == right:
                return

            mid = (left + right) // 2

            # Left child
            l = node * 2
            length = mid - left + 1
            sum2[l] += 2 * sum1[l] * lazy[node] + length * lazy[node] * lazy[node]
            sum1[l] += length * lazy[node]
            lazy[l] += lazy[node]

            # Right child
            r = node * 2 + 1
            length = right - mid
            sum2[r] += 2 * sum1[r] * lazy[node] + length * lazy[node] * lazy[node]
            sum1[r] += length * lazy[node]
            lazy[r] += lazy[node]

            lazy[node] = 0

        def update(node, left, right, ql, qr):
            if ql <= left and right <= qr:
                length = right - left + 1
                x = lazy[node]

                # Add 1 to every value in this segment
                sum2[node] += 2 * sum1[node] + length
                sum1[node] += length
                lazy[node] += 1
                return

            push(node, left, right)

            mid = (left + right) // 2

            if ql <= mid:
                update(node * 2, left, mid, ql, qr)

            if qr > mid:
                update(node * 2 + 1, mid + 1, right, ql, qr)

            # IMPORTANT: update parent after children change
            sum1[node] = sum1[node * 2] + sum1[node * 2 + 1]
            sum2[node] = sum2[node * 2] + sum2[node * 2 + 1]

        last = {}
        ans = 0

        for r, x in enumerate(nums):
            p = last.get(x, -1)

            update(1, 0, n - 1, p + 1, r)

            ans = (ans + sum2[1]) % MOD

            last[x] = r

        return ans
