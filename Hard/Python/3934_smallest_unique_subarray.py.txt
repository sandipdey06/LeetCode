class Solution(object):
    def smallestUniqueSubarray(self, nums):
        n = len(nums)
        for k in range(1, n + 1):
            count = {}
            for i in range(n-k+1):
                x = tuple(nums[i:i+k])
                count[x] = count.get(x, 0) + 1
            if 1 in count.values():
                return k
