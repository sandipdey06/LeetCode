class Solution {
    public int[] plusOne(int[] digits) {
        // digits[digits.length-1]+=1;
        // return digits;
        int a=digits.length;
        int []ans=new int[a+1];
        for (int i=a-1; i>=0; i--){
            if (digits[i]<9){
                digits[i]+=1;
                return digits;
            }
            digits[i]=0;
        }
        ans[0]=1;
        return ans;
    }
}
