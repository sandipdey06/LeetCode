
class Solution(object):
    def maxLength(self, nums):
        def gcd(a, b):
            return gcd(b, a % b) if b else a

        ans = 0
        for i in range(len(nums)):
            p, g, l = 1, 0, 1

            for j in range(i, len(nums)):
                p *= nums[j]
                g = gcd(g, nums[j])
                l = l * nums[j] // gcd(l, nums[j])

                if p == g * l:
                    ans = max(ans, j - i + 1)

        return ans
