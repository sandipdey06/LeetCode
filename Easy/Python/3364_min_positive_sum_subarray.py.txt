class Solution(object):
    def minimumSumSubarray(self, nums, l, r):
        ans = float('inf')
        for i in range(len(nums)):
            s = 0
            for j in range(i, min(len(nums), i+r)):
                s += nums[j]
                if j-i+1 >= l and s > 0:
                    ans = min(ans, s)
        return -1 if ans == float('inf') else ans
