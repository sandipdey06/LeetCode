class Solution(object):
    def mySqrt(self, x):
        # BINARY SEARCH
        l, r = 1, x
        ans = 0
        while l <= r:
            mid = (l + r) // 2
            if mid <= x // mid:
                ans = mid
                l = mid + 1
            else:
                r = mid - 1

        return ans
