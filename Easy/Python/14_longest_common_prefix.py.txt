class Solution(object):
    def longestCommonPrefix(self, strs):
        if strs[0] == "":
            return ""
        ans = ""
        #we are treating str[0]=flower as a base case to compare its elemnets
        for i in range(len(strs[0])):
            for s in strs:
                if i >= len(s) or s[i] != strs[0][i]:
                    return ans
            ans += strs[0][i]
        return ans
